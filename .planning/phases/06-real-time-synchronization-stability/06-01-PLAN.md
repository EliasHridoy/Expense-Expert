# Plan 06-01: Centralized SubscriptionManager & Subscription Lifecycle Hooks

## Overview
Implement a centralized, leak-free Firestore subscription manager (`RealtimeSyncManager`) and React subscription lifecycle hook (`useFirestoreSubscription`). This foundational infrastructure pools active Firestore `onSnapshot` listeners, applies reference counting to avoid duplicate subscriptions, and guarantees deterministic unsubscription when components unmount, when switching partitioned months, or upon user logout.

    ```typescript
    import { useEffect, useRef } from 'react';
    import { RealtimeSyncManager } from '../services/RealtimeSyncManager';
    import { SubscriptionOptions, Unsubscribe } from '../types/sync.types';

    export function useFirestoreSubscription(
      key: string | null | undefined,
      subscribeFn: () => Unsubscribe,
      deps: React.DependencyList = [],
      options: SubscriptionOptions = { enabled: true }
    ): void {
      const enabled = options.enabled !== false && Boolean(key);
      const subscribeFnRef = useRef(subscribeFn);
      subscribeFnRef.current = subscribeFn;

      useEffect(() => {
        if (!enabled || !key) return;

        const unsubscribe = RealtimeSyncManager.register(key, () => {
          return subscribeFnRef.current();
        });

        return () => {
          unsubscribe();
        };
      // eslint-disable-next-line react-hooks/exhaustive-deps
      }, [enabled, key, ...deps]);
    }
    ```
